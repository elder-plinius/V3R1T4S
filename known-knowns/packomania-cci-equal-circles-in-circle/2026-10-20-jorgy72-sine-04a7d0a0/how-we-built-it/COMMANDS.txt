# Exact build commands (from the ring-graft lens; paths relative to this package root)

# 1. Hexagonal truncation search (20k evaluations of the raw-truncation objective)
python3 how-we-built-it/a2search.py
# best raw truncation: theta=12.863 deg, s=0.02987721, offset (0.001694, 0.007549),
# 3949 smallest-|p| points -> margin -2.69e-05

# 2. Jam the truncation (overlap relaxation)
python3 how-we-built-it/jam.py
# -> jammed configuration at the raw radius

# 3. Smooth-max bottleneck polish with ratcheting (polish.py: L-BFGS-B on
#    logsumexp(alpha * overlaps) / alpha; after each round set r := feasible_r)
python3 how-we-built-it/deep_polish.py
# -> after 8 ratchets: margin -1.3e-08

# 4. Aggressive re-polish (12 outer rounds, alpha=1e6)
python3 how-we-built-it/batch_polish.py
python3 how-we-built-it/squeeze.py
# -> margin +2.372e-06 (50-digit certified)

# 5. Non-isometry vs the posted seed
python3 how-we-built-it/noniso.py object/candidate_n3949.txt
# -> NON-ISOMETRIC (max sorted-distance diff ~5e-03)

# 6. 50-digit certification

python3 verify-it-yourself/verify.py object/candidate_n3949.txt

# (or run the whole check with: sh verify-it-yourself/run_checks.sh)

# Notes on the build scripts
# - squeeze.py is included in how-we-built-it/ verbatim from the ring-graft run.
